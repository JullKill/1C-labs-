&НаКлиенте
Функция МассивВСтроку(Массив)
    Если Массив = Неопределено Или Массив.Количество() = 0 Тогда
        Возврат "Массив пустой или не создан";
    КонецЕсли;
   
    Результат = "";
    Для i = 0 По Массив.Количество() - 1 Цикл
        Если i > 0 Тогда
            Результат = Результат + ", ";
        КонецЕсли;
        Результат = Результат + Строка(Массив[i]);
    КонецЦикла;
    Возврат Результат;
КонецФункции

Процедура ОбратныйОбходТаблицы(Команда)
    Если СохраненныеДанныеТаблицы = Неопределено Или СохраненныеДанныеТаблицы.Количество() = 0 Тогда
        Сообщить("Сначала преобразуйте структуру в таблицу значений!");
        Возврат;
    КонецЕсли;
   
    Сообщить("Обратный обход таблицы (снизу вверх):");
    
    Индекс = СохраненныеДанныеТаблицы.Количество() - 1;
    Пока Индекс >= 0 Цикл
        Пара = СохраненныеДанныеТаблицы[Индекс];
        Сообщить(Пара.Параметр + ": " + Пара.Значение);
        Индекс = Индекс - 1;
    КонецЦикла;

&НаСервере
Функция СоздатьМассивНаСервере()
    
    НомерТемы = 10;
    ИнтервалОт = НомерТемы;           // 10
    ИнтервалДо = НомерТемы + 10;       // 20
    КоличествоЭлементов = НомерТемы + 50; // 60
    
    Массив = Новый Массив;
    Генератор = Новый ГенераторСлучайныхЧисел();
    
    Для і = 1 По КоличествоЭлементов Цикл
        СлучайноеЧисло = Генератор.СлучайноеЧисло(ИнтервалОт, ИнтервалДо);
        Массив.Добавить(СлучайноеЧисло);
    КонецЦикла;
    
    Возврат Массив;
    
КонецФункции

&НаСервере
Функция УдалитьДублиНаСервере(ВходнойМассив)
    
    Если ВходнойМассив = Неопределено Тогда
        Возврат Неопределено;
    КонецЕсли;
   
    ТаблицаУникальных = Новый ТаблицаЗначений;
    ТаблицаУникальных.Колонки.Добавить("Число");
   
    Для каждого Число Из ВходнойМассив Цикл
        Найдено = Ложь;
        Для каждого СтрокаТаблицы Из ТаблицаУникальных Цикл
            Если СтрокаТаблицы.Число = Число Тогда
                Найдено = Истина;
                Прервать;
            КонецЕсли;
        КонецЦикла;
       
        Если Не Найдено Тогда
            НоваяСтрока = ТаблицаУникальных.Добавить();
            НоваяСтрока.Число = Число;
        КонецЕсли;
    КонецЦикла;
   
    НовыйМассив = Новый Массив;
    Для каждого СтрокаТаблицы Из ТаблицаУникальных Цикл
        НовыйМассив.Добавить(СтрокаТаблицы.Число);
    КонецЦикла;
   
    Возврат НовыйМассив;
    
КонецФункции

Функция СортироватьМассивНаСервере(ВходнойМассив)
    
    Если ВходнойМассив = Неопределено Тогда
        Возврат Неопределено;
    КонецЕсли;
   
    ТаблицаДляСортировки = Новый ТаблицаЗначений;
    ТаблицаДляСортировки.Колонки.Добавить("Число");
   
    Для каждого Число Из ВходнойМассив Цикл
        Строка = ТаблицаДляСортировки.Добавить();
        Строка.Число = Число;
    КонецЦикла;
   
    ТаблицаДляСортировки.Сортировать("Число Возр");
   
    НовыйМассив = Новый Массив;
    Для каждого Строка Из ТаблицаДляСортировки Цикл
        НовыйМассив.Добавить(Строка.Число);
    КонецЦикла;
   
    Возврат НовыйМассив;
    
КонецФункции

&НаСервере
Функция ПолучитьДанныеТаблицыНаСервере(Структура)
    
    Данные = Новый Массив;
   
    МассивКлючей = Новый Массив;
    МассивКлючей.Добавить("Наименование");
    МассивКлючей.Добавить("Вес");
    МассивКлючей.Добавить("Цена");
    МассивКлючей.Добавить("Состав");
   
    Для каждого Ключ Из МассивКлючей Цикл
        ПараДанных = Новый Структура;
        ПараДанных.Вставить("Параметр", Ключ);
        ПараДанных.Вставить("Значение", Структура[Ключ]);
        Данные.Добавить(ПараДанных);
    КонецЦикла;
   
    Возврат Данные;
    
КонецФункции
